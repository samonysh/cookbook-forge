// ilm-template.typ.txt
//
// 基于 ilm 模板的优化版 Typst 主文件模板。
// 样式优化参考 graph-notes 项目的排版方案：
//   - 中文字体：LXGW WenKai（霞鹜文楷）+ 回退字体栈
//   - 代码字体：LXGW WenKai Mono + 回退
//   - 行内代码：浅灰背景 + 圆角
//   - 块级代码：浅灰背景 + 灰色边框 + 圆角
//   - Callout 提示框：四色（蓝/橙/红/绿）左边框 + 浅色背景
//   - 表格：booktabs 风格（粗顶线 + 细中线 + 粗底线）
//   - 图片：70% 宽度居中 + caption
//   - 标题自动编号：第 C 章 / C.S（模板编号，章节文件不得再写手动编号）
//   - 图表自动编号：图 C-S / 表 C-S（每章重置，正文用 @fig-C-S / @tab-C-S 引用）
//
// 配置驱动（P1-4）：字体、编号格式、supplement 均由 assets/typst-default.config.json
// 生成，lmdx2typst.mjs 读取本模板后替换 __PLACEHOLDER__ 占位符。
// 后缀 .typ.txt 以兼容 SkillHub 文件类型白名单，脚本读取时按纯文本处理。

#import "@preview/ilm:2.1.1": *

// ---------- 中文字体配置（__FONT_*__ 由配置生成） ----------
// 注意：字体栈回退项必须是字符串（"serif"/"monospace"），裸写 serif/monospace
// 在 typst 0.15+ 会报 unknown variable 编译错误。
#set text(
  lang: "zh",
  font: (__FONT_TEXT_ARGS__),
  size: __FONT_TEXT_SIZE__,
)

// ---------- ilm 模板配置 ----------
#show: ilm.with(
  title: [__BOOK_TITLE__],
  authors: "__BOOK_AUTHOR__",
  date: datetime.today(),
  abstract: [__BOOK_ABSTRACT__],
  bibliography: none,
  figure-index: (enabled: true, title: "插图索引"),
  table-index: (enabled: true, title: "表格索引"),
  listing-index: (enabled: true, title: "代码索引"),
)

// ---------- 标题自动编号（配置驱动，P0-1） ----------
// 模板自动编号「第 C 章 / C.S」，章节文件中的手动编号会被
// scripts/lib/typst-optimize.mjs 在组装期剥离，避免双重编号。
__HEADING_NUMBERING__

// ---------- 图表编号与 supplement（配置驱动，P0-2） ----------
// 图/表按「章-序」编号（每章重置），正文引用用 @fig-C-S / @tab-C-S。
// lmdx2typst.mjs / typst-check.mjs --fix 会自动注入 label 并转换纯文本引用。
__FIGURE_NUMBERING__

// ---------- 长表格跨页（P0 表格质量） ----------
// figure 默认不可断页，内部长表格会溢出页面；官方推荐用 show rule 打开 figure 的
// 可断性（https://typst.app/docs/reference/model/figure/）。只作用于表格 figure，
// 图片 figure 不受影响；超阈值（tables.splitThreshold）的表格由优化器另注入
// breakable: true，两处配合实现长表跨页续排。
#show figure.where(kind: table): set block(breakable: true)

// ---------- 代码块样式优化 ----------
// 行内代码：浅灰背景 + 小圆角
#show raw.where(block: false): box.with(
  fill: luma(240),
  inset: (x: 3pt, y: 0pt),
  outset: (y: 3pt),
  radius: 2pt,
)

// 块级代码：浅灰背景 + 灰色边框 + 圆角 + 内边距
#show raw.where(block: true): block.with(
  fill: luma(245),
  stroke: 1pt + rgb("#444"),
  inset: 10pt,
  radius: 4pt,
)

// 代码字体
#show raw: set text(
  font: (__FONT_CODE_ARGS__),
  size: __FONT_CODE_SIZE__,
)

// ---------- Callout 提示框 ----------
// 定义已拆分到独立的 callout.typ 模块：Typst 的 #include 不继承本文件的 #let 词法作用域，
// 章节文件必须自行 `#import "../callout.typ": callout-box`（lmdx2typst.mjs 会生成该文件）。
// 四色：蓝（本章地图）、橙（Recipe）、红（陷阱）、绿（核心要点）。

// ---------- 章节内容 ----------
// 各章节通过 #include 引入
